cmake_minimum_required(VERSION 3.21)
project(omawser_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets WebEngineWidgets Test Network)
add_library(oma_core core.cpp)
target_link_libraries(oma_core PUBLIC Qt6::Core)
add_library(oma_window browser.cpp)
target_link_libraries(oma_window PUBLIC oma_core Qt6::Widgets Qt6::WebEngineWidgets)
add_executable(omawser-qt main.cpp)
target_link_libraries(omawser-qt PRIVATE oma_window)
add_executable(core-test core_test.cpp)
target_link_libraries(core-test PRIVATE oma_core Qt6::Test)
add_executable(gui-test gui_test.cpp)
target_link_libraries(gui-test PRIVATE oma_window Qt6::Test Qt6::Network)
enable_testing()
add_test(NAME core COMMAND core-test)
add_test(NAME version COMMAND omawser-qt --version)
set_tests_properties(version PROPERTIES PASS_REGULAR_EXPRESSION "Chromium [0-9]+\\.[0-9]+")
# GUI tests require a real display and explicitly isolated XDG paths; run manually.
